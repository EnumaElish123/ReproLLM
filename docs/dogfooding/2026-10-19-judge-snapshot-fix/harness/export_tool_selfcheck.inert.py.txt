 records=[]

  folder=OUT/label;folder.mkdir();md=folder/'statement.md';md.write_text(body);report=folder/'report.json';args=argv+['--markdown',str(md),'--report',str(report)]

 summary={'schema_version':1,'status':'PASS','new_product_commands':0,'gpu_or_api_calls':0,'positive_cases':1,'negative_cases':len(records)-1,'records':records,'checker_sha256':sha((HERE/'verify_export.py').read_bytes()),'retained_source_candidate':commit,'scope':'Read-only verifier tests over already retained DP export, with disposable Markdown/binding/upstream mutations only; no new native/current-candidate acceptance.'}
