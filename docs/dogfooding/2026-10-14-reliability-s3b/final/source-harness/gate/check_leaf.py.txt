def load(raw):
    def unique(pairs):
        result={}
        for key,value in pairs:
            if key in result:raise ValueError('duplicateJSONkey')
            result[key]=value
        return result
    def finite(s):raise ValueError('nonfiniteJSON')
    return json.loads(raw.decode('utf-8','strict'),object_pairs_hook=unique,parse_constant=finite)


def equal(a,b):
    if type(a) is not type(b):return False
    if isinstance(a,dict):return a.keys()==b.keys() and all(equal(a[k],b[k]) for k in a)
    if isinstance(a,list):return len(a)==len(b) and all(equal(x,y) for x,y in zip(a,b))
    return a==b


def phase(subject,when):return {'subject':subject,'phase':when,'verification':'unverified'}

def leaf(value,source='run',detail=None,context=None):
    return {'value':value,'source':source,'detail':detail,'confidence':'observed' if source=='run' else 'declared','context':context,'alternatives':[]}

def context(path):
    if path=='environment.python' or path.startswith('environment.packages.'):return phase('recorder','post_child')
    if path=='environment.env_capture' or path.startswith('environment.env.'):return phase('inherited_environment','pre_child')
    if path=='environment.hostname_sha256':return phase('host','pre_child')
    if path.startswith(('environment.','hardware.')):return phase('host','post_child')
    return phase('recorder','pre_child')


def flatten_values(data,prefix=''):
    result={}
    if isinstance(data,dict):
        for key,value in sorted(data.items()):result.update(flatten_values(value,prefix+'.'+key if prefix else key))
    elif data is not None:result[prefix]=data
    return result


def expected_state(f,which):
    """Case-specific uncached projection, including all evidence alternatives."""
    record=f['run_records'][which];locked=f['validated_lock'];groups={}
    def add(path,item):
        if item['source'] in {'manifest','lock'} and item['context'] is None:item['context']=phase('input_file','pre_child')
        groups.setdefault(path,[]).append(item)
    for path,value in {'generation.temperature':0.1,'execution.command':'python r04-child.py --config r04-config.json','execution.cwd':'.','execution.seed':17,'execution.config_files':['r04-config.json']}.items():add(path,leaf(value,'manifest'))
    add('generation.temperature',leaf(0.1,'lock'))
    env=locked['environment']
    lock_values=flatten_values({key:env[key] for key in ('python','platform','packages')},'environment')
    lock_values.update(flatten_values(env['gpu'],'hardware'))
    for path,value in lock_values.items():
        subject='recorder' if path=='environment.python' or path.startswith('environment.packages.') else 'host'
        add(path,leaf(value,'lock',context=phase(subject,'lock_resolution')))
    for file in locked['files']:add('files.'+file['path']+'.sha256',leaf(file['sha256'],'lock'))
    for section in ['code','environment','command','run_id','started_at','ended_at','duration_seconds']:
        for path,value in flatten_values(record[section],section).items():
            add(path,leaf(value,context=context(path) if path.startswith(('environment.','command.')) else None))
    hardware={key:value for key,value in record['hardware'].items() if key!='gpus'}
    hardware['gpus']={str(gpu['index']):gpu for gpu in record['hardware']['gpus']}
    for path,value in flatten_values(hardware,'hardware').items():add(path,leaf(value,context=context(path)))
    for file in record['files']:add('files.'+file['path']+'.sha256',leaf(file['sha256'],context=phase('input_file','pre_child')))
    value=0.2 if which=='A' else 0.1
    primary=leaf(value,detail='cli:--temperature' if which=='A' else 'config:r04-config.json:temperature',context=phase('recorder' if which=='A' else 'input_file','pre_child'))
    if which=='A':primary['alternatives']=[leaf(0.1,detail='cli:--temperature',context=phase('recorder','pre_child')),leaf(0.1,detail='config:r04-config.json:temperature',context=phase('input_file','pre_child'))]
    add('generation.temperature',primary)
    def rank(x):return {'cli':9,'config':8,'env':7}.get((x['detail'] or '').split(':')[0],6) if x['source']=='run' else {'working_tree':5,'lock':4,'manifest':3,'default':1}[x['source']]
    def order(x):return (-rank(x),x['detail'] or '',json.dumps(x['value'],sort_keys=True,ensure_ascii=False))
    flat={}
    for path,items in sorted(groups.items()):
        chosen=min(items,key=order);other=[]
        for item in items:
            if item is not chosen:other.append({**item,'alternatives':[]})
            other.extend(item['alternatives'])
        flat[path]={**chosen,'alternatives':sorted(other,key=order)}
    state={key:{} for key in STATE_BRANCHES}
    state.update({key:None for key in ['run_id','started_at','ended_at','duration_seconds']},profiles=['validation'])
    for path,item in flat.items():
        parts=['files',path[6:-7],'sha256'] if path.startswith('files.') and path.endswith('.sha256') else path.split('.')
        branch=state
        for part in parts[:-1]:branch=branch.setdefault(part,{})
        branch[parts[-1]]=item
    return {'schema_version':1,'state':state,'flat':flat,'roundtrip_equal':True,'overlap_merge_equal':True}


def full_diff(f,which_a='A',which_b='B'):
    left=expected_state(f,which_a)['flat'];right=expected_state(f,which_b)['flat'];changes=[];same=0
    for path in sorted(left.keys()|right.keys()):
        before,after=left.get(path),right.get(path)
        if before and after and before['value']==after['value']:same+=1;continue
        a=before['value'] if before else None;b=after['value'] if after else None
        severity='HIGH' if path=='generation.temperature' else 'NONE' if path in {'run_id','started_at','ended_at','duration_seconds','environment.hostname_sha256'} else 'MEDIUM'
        notes=[]
        for side,item in [('a',before),('b',after)]:
            if item is None:continue
            if any(other['value']!=item['value'] for other in item['alternatives']):notes.append(side+' has inconsistent sources (see audit)')
            if item['context']:
                c=item['context'];notes.append(f"{side}: {c['subject']}/{c['phase']}; child unverified")
        changes.append({'path':path,'status':'added' if before is None else 'removed' if after is None else 'changed','a':a,'b':b,'severity':severity,'note':'; '.join(notes) or None})
    def section(path):
        head=path.split('.')[0];return (SECTIONS.index(head) if head in SECTIONS else len(SECTIONS),head)
    changes.sort(key=lambda c:(-SEVERITIES.index(c['severity']),section(c['path']),c['path']))
    counts={severity:sum(c['severity']==severity for c in changes) for severity in SEVERITIES}
    highest=next((severity for severity in reversed(SEVERITIES) if counts[severity]),'NONE')
    return {'schema_version':1,'reprollm_version':'0.7.0','a':{'kind':'run','ref':f['run_records'][which_a]['run_id']},'b':{'kind':'run','ref':f['run_records'][which_b]['run_id']},'changes':changes,'summary':{'counts':counts,'highest':highest,'same':same},'filtered_below':None}


def diff_text(report):
    lines=[f"a: run {report['a']['ref']}",f"b: run {report['b']['ref']}",'Child environment and input consumption are unverified.']
    def value(v):
        text=v if isinstance(v,str) else json.dumps(v,sort_keys=True,ensure_ascii=False)
        text=text.replace('\n',r'\n').replace('\r',r'\r')
        return re.sub(r'(?<![A-Za-z0-9])(?:sha256:)?[a-fA-F0-9]{40,64}(?![A-Za-z0-9])',lambda m:('sha256:' if m[0].startswith('sha256:') else '')+m[0].removeprefix('sha256:')[:12],text)
    groups={}
    for change in report['changes']:groups.setdefault(change['path'].split('.')[0],[]).append(change)
    for section in sorted(groups,key=lambda s:(SECTIONS.index(s) if s in SECTIONS else len(SECTIONS),s)):
        lines+=['',section]
        for change in groups[section]:
            lines.append(f"  {change['path']}  {value(change['a'])} → {value(change['b'])}  [{change['severity']}]")
            if change['note']:lines.append('    '+change['note'])
    severity=report['summary']['highest'];count=report['summary']['counts'][severity]
    verdict=f'Highest drift: {severity} ({count} changes). These runs are not directly comparable.' if severity=='HIGH' else f'Highest drift: {severity} ({count} changes). Results may differ; review the changes above.' if severity in {'MEDIUM','MEDIUM_HIGH'} else 'No reproducibility-relevant drift in recorded values.'
    return ('\n'.join(lines+['',verdict])+'\n').encode()


def bind(document,values):
    if isinstance(document,str):
        for token,value in values.items():document=document.replace('${'+token+'}',value)
        assert '${' not in document
        return document
    if isinstance(document,list):return [bind(item,values) for item in document]
    if isinstance(document,dict):return {key:bind(item,values) for key,item in document.items()}
    return document


def audit_expected(name,which,f,timestamp):
    raw=ORACLE.read_bytes();assert sha(raw)==ORACLE_SHA
    template=load(raw)['complete_audit_templates'][name][which]
    assert template['final_message_binding_pending'] is False
    values={'FIXTURE.manifest_sha256':fingerprint(f['input_before']['reprollm.yaml'])['sha256'],
            'FIXTURE.lock_sha256':fingerprint(f['input_before']['reprollm.lock'])['sha256'],
            'AUDIT.generated_at':timestamp}
    for case in f['run_records']:values[case+'.run_path']='.reprollm/runs/'+f['run_records'][case]['run_id']+'/run.json'
    expected=bind(template['report_template'],values)

        for package,version in f['metadata']['packages'].items():
            finding['message']=finding['message'].replace(package+'==<version>',package+'=='+version)
            finding['fix_hint']=finding['fix_hint'].replace(package+'==<version>',package+'=='+version)
    return expected


def timestamp_valid(value,result):
    if not isinstance(value,str) or not re.fullmatch(r'\d{4}-\d\d-\d\dT\d\d:\d\d:\d\dZ',value):return False
    stamp=datetime.fromisoformat(value.replace('Z','+00:00')).timestamp()
    return result['record']['started_epoch']-1<=stamp<=result['record']['ended_epoch']+1

class Code(str):pass

def display(value):
    if isinstance(value,Code):return str(value)
    text=html.escape(str(value),quote=False)
    for original,replacement in [('\\','\\\\'),('|','\\|'),('`','&#96;'),('[','&#91;'),(']','&#93;'),('*','&#42;'),('_','&#95;'),('~','&#126;'),('\r\n','<br>'),('\r','<br>'),('\n','<br>')]:text=text.replace(original,replacement)
    return text

def code(value):
    text=str(value).replace('\n',' ').replace('\r',' ');assert '`' not in text
    return Code('`'+text+'`')
