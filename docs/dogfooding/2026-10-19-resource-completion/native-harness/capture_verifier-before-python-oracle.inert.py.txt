"""Read-only acceptance checks for the five approved, bounded native replay cases.

No model execution, HTTP, installation, source mutation or evidence replacement.
Original Git bytes are the input oracle; State is used only for the current
effective-binding contract, never to invent expected experimental values.
"""

from __future__ import annotations

import getpass
import hashlib
import json
import math
import re
import socket
import subprocess
import sys
from collections.abc import Callable
from functools import wraps
from pathlib import Path, PureWindowsPath
from typing import Any, ParamSpec, TypeVar

import yaml

from reprollm.run.contextual_text import redact_contextual_text
from reprollm.schemas.diff_report import DiffReport
from reprollm.schemas.finding import AuditReport
from reprollm.schemas.manifest import Manifest
from reprollm.schemas.run_record import RunRecord
from reprollm.schemas.state import State

_LOCAL_PATH = re.compile(
    r"(?i)\bfile:[/\\]{2}"
    r"|(?<![\w])[A-Za-z]:[\\/](?=[^\s\"'<>;,\)\]}>])"
    r"|(?<![\w:/\\*<.])/(?=[^\s\"'<>;,\)\]}>])[^\s\"'<>;,\)\]}>]*"
    r"|\\\\[A-Za-z0-9_.-]+\\+[A-Za-z0-9_.-]+"
)

_P = ParamSpec("_P")
_T = TypeVar("_T")

_COMMIT = re.compile(r"[0-9a-f]{40}\Z")
_NONE = {"run_id", "started_at", "ended_at", "duration_seconds", "environment.hostname_sha256"}


class VerificationError(AssertionError):
    """A fixed diagnostic that never includes untrusted values or exception text."""


def _safe_failure(fn: Callable[_P, _T]) -> Callable[_P, _T]:
    @wraps(fn)
    def wrapped(*args: _P.args, **kwargs: _P.kwargs) -> _T:
        try:
            return fn(*args, **kwargs)
        except VerificationError:
            raise
        except Exception:
            raise VerificationError("verification input or filesystem invalid") from None

    return wrapped


def _require(ok: bool, reason: str) -> None:
    if not ok:
        raise VerificationError(reason)


def _digest(raw: bytes) -> str:
    return "sha256:" + hashlib.sha256(raw).hexdigest()


def _equal(a: Any, b: Any) -> bool:
    def normal(v: Any) -> Any:
        if isinstance(v, str):
            if v.lower() in ("true", "false"):
                return v.lower() == "true"
            try:
                return float(v) if "." in v or "e" in v.lower() else int(v)
            except ValueError:
                return v
        if isinstance(v, list):
            return [normal(item) for item in v]
        return v

    left, right = normal(a), normal(b)
    if isinstance(left, list) and isinstance(right, list):
        return len(left) == len(right) and all(
            _equal(x, y) for x, y in zip(left, right, strict=True)
        )
    if isinstance(left, (int, float)) and isinstance(right, (int, float)):
        return math.isclose(left, right, rel_tol=1e-9)
    return bool(left == right)


def _inside(root: Path, relative: str) -> Path:
    _require(
        bool(relative)
        and "\x00" not in relative
        and not Path(relative).is_absolute()
        and not PureWindowsPath(relative).anchor
        and ".." not in Path(relative).parts,
        "invalid evidence location",
    )
    result = root / relative
    _require(not result.is_symlink(), "symlink evidence is unsupported")
    _require(result.resolve().is_relative_to(root.resolve()), "evidence escapes its root")
    return result


def _git(case: Path, *argv: str) -> bytes:
    result = subprocess.run(
        ["git", "-C", str(case), *argv], capture_output=True, check=False, timeout=30
    )
    _require(result.returncode == 0, "Git input evidence unavailable")
    return result.stdout


def _raw_input(case: Path, commit: str, relative: str) -> bytes:
    _inside(case, relative)
    _require(bool(_COMMIT.fullmatch(commit)), "invalid captured commit")
    return _git(case, "show", commit + ":" + relative)


def _parse(raw: bytes, suffix: str) -> Any:
    text = raw.decode("utf-8")
    if suffix == ".json":
        return json.loads(text)
    if suffix == ".toml":
        if sys.version_info >= (3, 11):
            import tomllib
        else:
            import tomli as tomllib
        return tomllib.loads(text)
    return yaml.safe_load(text)


def _read_run(run_dir: Path) -> RunRecord:
    try:
        return RunRecord.model_validate_json(_inside(run_dir, "run.json").read_bytes())
    except Exception:
        raise VerificationError("invalid typed run record") from None


def _privacy(root: Path, text: str) -> None:
    _require(_LOCAL_PATH.search(text) is None, "persisted absolute path or local URI")
    for identity in (socket.gethostname(), getpass.getuser()):
        if identity:
            _require(
                re.search(r"(?<![\w])" + re.escape(identity) + r"(?![\w])", text) is None,
                "persisted machine identity",
            )
    try:
        # Independent paths above avoid the known RunPrivacy slash false positive.
        # This pass checks only the existing supported secret-pattern context.
        safe, _ = redact_contextual_text(text)
    except Exception:
        raise VerificationError("unsafe or unverifiable persisted text") from None
    _require(safe == text, "persisted supported secret or unsafe text")


def _semantic_copy(original: Any, snapshot: Any, case: Path) -> None:
    if isinstance(original, dict):
        _require(
            isinstance(snapshot, dict) and original.keys() == snapshot.keys(),
            "snapshot keys changed",
        )
        for key, value in original.items():
            _semantic_copy(value, snapshot[key], case)
    elif isinstance(original, list):
        _require(
            isinstance(snapshot, list) and len(original) == len(snapshot), "snapshot list changed"
        )
        for before, after in zip(original, snapshot, strict=True):
            _semantic_copy(before, after, case)
    elif original != snapshot:
        _require(isinstance(original, str) and isinstance(snapshot, str), "snapshot scalar changed")
        # Only independently recognizable whole absolute-path scalars may change.
        # Stop tokens, globs, URLs, keys, model identities and bindings stay exact.
        match = _LOCAL_PATH.fullmatch(original)
        _require(match is not None, "non-secret snapshot scalar changed")
        expected = "<REDACTED:path>"
        try:
            relative = Path(original).relative_to(case.resolve())
            if ".." not in relative.parts:
                expected = relative.as_posix()
        except ValueError:
            pass
        _require(snapshot == expected, "absolute-path snapshot substitution incorrect")


def _snapshot(case: Path, run_dir: Path, relative: str, snapshot: str, raw: bytes) -> None:
    saved = _inside(run_dir, snapshot).read_bytes()
    _privacy(case, saved.decode("utf-8"))
    if saved != raw:
        try:
            _semantic_copy(
                _parse(raw, Path(relative).suffix), _parse(saved, Path(relative).suffix), case
            )
        except VerificationError:
            raise
        except Exception:
            raise VerificationError("changed snapshot cannot be checked structurally") from None


def _binding_sources(manifest: Manifest, field: str) -> set[tuple[str, str | None, str]]:
    binding = manifest.bindings.get(field)
    _require(binding is not None, "expected field binding undeclared")
    assert binding is not None
    result: set[tuple[str, str | None, str]] = set()
    if binding.config:
        path, _, key = binding.config.partition(":")
        result.add(("config", path, key))
    if binding.cli:
        result.add(("cli", None, binding.cli))
    if binding.env:
        result.add(("env", None, binding.env))
    return result


def _effective(run_dir: Path, field: str, expected: Any) -> dict[str, Any]:
    run = _read_run(run_dir)
    _require(run.manifest is not None, "manifest capture missing")
    assert run.manifest is not None
    manifest = Manifest.model_validate(
        _parse(_inside(run_dir, run.manifest.snapshot).read_bytes(), ".yaml")
    )
    sources = _binding_sources(manifest, field)
    observations = run.bindings_observed.get(field, [])
    _require(bool(observations), "expected binding observation missing")
    _require(
        all(
            (item.source.type, item.source.path, item.source.key) in sources
            for item in observations
        ),
        "binding observation source differs from declaration",
    )
    leaf = State.from_run(run, run_dir=run_dir).flatten().get(field)
    _require(
        leaf is not None and leaf.source == "run" and _equal(leaf.value, expected),
        "effective binding value mismatch",
    )
    assert leaf is not None
    details = {f"{kind}:{path + ':' if path else ''}{key}" for kind, path, key in sources}
    _require(leaf.detail in details, "effective binding source mismatch")
    return {
        "field": field,
        "value": leaf.value,
        "source": leaf.detail,
        "observations": len(observations),
    }


@_safe_failure
def verify_capture(
    case: Path,
    run_dir: Path,
    expected_field: str,
    expected_value: Any,
    expected_argv: list[str],
    expected_packages: dict[str, str],
    expected_code_commit: str,
    require_gpu: bool = True,
) -> dict[str, Any]:
    """Check a fresh capture before the orchestrator moves the native outputs."""
    run = _read_run(run_dir)
    _require(
        run.status.value == "completed" and run.exit_code == 0,
        "native child did not complete successfully",
    )
    _require(not run.warnings, "required capture has warnings or missing data")
    _require(run.command.argv == expected_argv, "captured command mismatch")
    _require(run.command.cwd == ".", "captured working directory mismatch")
    _require(
        run.code.commit == expected_code_commit and not run.code.dirty,
        "captured code identity mismatch",
    )
    _require(not _git(case, "status", "--porcelain"), "native execution changed tracked inputs")
    _require(
        run.environment is not None and run.hardware is not None,
        "full environment or hardware capture missing",
    )
    assert run.environment is not None
    assert run.hardware is not None
    _require(
        not require_gpu or (run.hardware.source == "nvidia_smi" and any(gpu.index == 0 for gpu in run.hardware.gpus)),
        "approved GPU hardware evidence missing",
    )
    for attribute, prefix in (("slurm", "SLURM_"), ("pbs", "PBS_"), ("lsf", "LSB_")):
        expected_namespace = {
            key: value
            for key, value in run.environment.env.items()
            if key.startswith(prefix) and isinstance(value, str)
        }
        captured_namespace = getattr(run.scheduler, attribute) if run.scheduler else None
        _require(
            captured_namespace == (expected_namespace or None),
            "scheduler namespace differs from captured environment metadata",
        )
    _require(run.environment.packages == expected_packages, "launcher package metadata mismatch")
    _require(
        not require_gpu or run.environment.env.get("CUDA_VISIBLE_DEVICES") == "0",
        "approved GPU selection not captured",
    )
    _require(
        run.manifest is not None
        and run.lock is not None
        and bool(run.files)
        and bool(run.artifacts),
        "required documents, inputs or outputs missing",
    )
    checked_inputs: list[str] = []
    for entry in run.files:
        raw = _raw_input(case, expected_code_commit, entry.path)
        _require(
            _digest(raw) == entry.sha256 and len(raw) == entry.size_bytes,
            "captured input hash or size mismatch",
        )
        _require(_inside(case, entry.path).read_bytes() == raw, "input changed during execution")
        text_eligible = len(raw) <= 1048576 and b"\0" not in raw
        try:
            raw.decode("utf-8")
        except UnicodeDecodeError:
            text_eligible = False
        if text_eligible:
            _require(entry.snapshot is not None, "eligible input snapshot missing")
        if entry.snapshot is not None:
            _snapshot(case, run_dir, entry.path, entry.snapshot, raw)
            if not entry.redacted:
                _require(
                    _inside(run_dir, entry.snapshot).read_bytes() == raw,
                    "unredacted snapshot differs from source",
                )
        checked_inputs.append(entry.path)
    for document in (run.manifest, run.lock):
        assert document is not None
        raw = _raw_input(case, expected_code_commit, document.path)
        _require(_digest(raw) == document.sha256, "document source hash mismatch")
        _require(
            _inside(case, document.path).read_bytes() == raw, "document changed during execution"
        )
        _snapshot(case, run_dir, document.path, document.snapshot, raw)
    assert run.manifest is not None
    manifest = Manifest.model_validate(
        _parse(_inside(run_dir, run.manifest.snapshot).read_bytes(), ".yaml")
    )
    declared: set[str] = set(manifest.execution.config_files or []) if manifest.execution else set()
    declared.update(
        item.config.partition(":")[0] for item in manifest.bindings.values() if item.config
    )
    declared.update(item.path for item in manifest.prompts.values() if item.path)
    _require(declared <= set(checked_inputs), "declared input capture incomplete")
    patterns = manifest.artifacts.outputs or [] if manifest.artifacts else []
    expected_outputs = {
        p.relative_to(case).as_posix()
        for pattern in patterns
        for p in case.glob(pattern)
        if p.is_file()
    }
    _require(
        expected_outputs == {item.path for item in run.artifacts},
        "declared output capture incomplete",
    )
    for artifact in run.artifacts:
        raw_path = _inside(case, artifact.path)
        digest = hashlib.sha256()
        with raw_path.open("rb") as stream:
            for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
                digest.update(block)
        _require(
            "sha256:" + digest.hexdigest() == artifact.sha256
            and raw_path.stat().st_size == artifact.size_bytes,
            "output hash or size mismatch",
        )
    _require(
        all(_inside(run_dir, name).is_file() for name in ("stdout.log", "stderr.log")),
        "required saved output log missing",
    )
    checked_saved = 0
    for path in sorted(run_dir.rglob("*")):
        _require(not path.is_symlink(), "symlink in owned run directory")
        if path.is_file():
            _privacy(case, path.read_bytes().decode("utf-8"))
            checked_saved += 1
    _require(run.schema_version == 2 and run.input_capture is not None, "current run2 capture missing")
    capture = run.input_capture
    _require(capture.baseline_phase == "pre_child" and capture.check_phase == "post_child" and capture.consumption == "unverified", "run2 input phases invalid")
    _require(bool(capture.checks) and all(c.change == "unchanged" and c.before.sha256 == c.after.sha256 for c in capture.checks), "native inputs changed during child execution")
    effective = _effective(run_dir, expected_field, expected_value)
    return {
        "schema_version": 1,
        "status": "PASS",
        "run_id": run.run_id,
        "effective_binding": effective,
        "input_files_checked": len(checked_inputs),
        "artifacts_checked": len(run.artifacts),
        "saved_files_privacy_checked": checked_saved,
        "source_hashes_are_original_bytes": True,
        "environment_proof": (
            "same launcher environment metadata; "
            "external child interpreter is not independently attested"
        ),
    }


def _report(value: dict[str, Any] | Path | str) -> dict[str, Any]:
    if isinstance(value, dict):
        return value
    result: dict[str, Any] = json.loads(value.read_text() if isinstance(value, Path) else value)
    _require(isinstance(result, dict), "report root is not an object")
    return result


@_safe_failure
def verify_pair(
    case: Path,
    a_run_dir: Path,
    b_run_dir: Path,
    field: str,
    values: tuple[Any, Any],
    severity: str,
    config_path: str,
    pair_json: dict[str, Any] | Path | str,
    self_json: dict[str, Any] | Path | str,
    audit_a_json: dict[str, Any] | Path | str,
    audit_b_json: dict[str, Any] | Path | str,
) -> dict[str, Any]:
    """Verify the target gold and report every other current semantic delta."""
    left_run, right_run = _read_run(a_run_dir), _read_run(b_run_dir)
    _require(
        all(
            run.status.value == "completed" and run.exit_code == 0 and not run.warnings
            for run in (left_run, right_run)
        ),
        "paired native execution or capture incomplete",
    )
    effective = [
        _effective(folder, field, expected)
        for folder, expected in zip((a_run_dir, b_run_dir), values, strict=True)
    ]
    _require(
        left_run.environment is not None and right_run.environment is not None,
        "pair environment missing",
    )
    assert left_run.environment is not None and right_run.environment is not None
    _require(
        left_run.environment.packages == right_run.environment.packages,
        "paired package environment changed",
    )
    documents = [_report(item) for item in (pair_json, self_json, audit_a_json, audit_b_json)]
    for document in documents:
        _privacy(case, json.dumps(document, ensure_ascii=False))
    pair, self_report = (DiffReport.model_validate(item) for item in documents[:2])
    audits = [AuditReport.model_validate(item) for item in documents[2:]]
    _require(
        pair.filtered_below is None and self_report.filtered_below is None,
        "filtered diff cannot prove full comparison",
    )
    _require(
        pair.a.ref == left_run.run_id and pair.b.ref == right_run.run_id, "pair sources mismatch"
    )
    _require(
        self_report.a.ref == left_run.run_id
        and self_report.b.ref == left_run.run_id
        and not self_report.changes,
        "self diff is not empty or source mismatch",
    )
    left = State.from_run(left_run, run_dir=a_run_dir).flatten()
    right = State.from_run(right_run, run_dir=b_run_dir).flatten()
    expected_paths = {
        key
        for key in left.keys() | right.keys()
        if key not in left or key not in right or not _equal(left[key].value, right[key].value)
    }
    _require(
        len(pair.changes) == len({item.path for item in pair.changes})
        and {item.path for item in pair.changes} == expected_paths,
        "full diff fields missing or duplicated",
    )
    explained: list[dict[str, Any]] = []
    extras: list[dict[str, Any]] = []
    for change in pair.changes:
        before, after = left.get(change.path), right.get(change.path)
        _require(
            _equal(change.a, before.value if before else None)
            and _equal(change.b, after.value if after else None),
            "diff value differs from effective State",
        )
        expected_status = "added" if before is None else "removed" if after is None else "changed"
        _require(change.status == expected_status, "diff change status mismatch")
        reason = "requires independent explanation"
        expected_severity = None
        if change.path == field:
            expected_severity, reason = severity, "the predeclared experimental parameter"
            _require(
                _equal(change.a, values[0]) and _equal(change.b, values[1]),
                "target drift values mismatch",
            )
        elif change.path == f"files.{config_path}.sha256":
            expected_severity, reason = "HIGH", "original configuration bytes changed"
        elif change.path == "code.commit":
            expected_severity, reason = "MEDIUM", "clean disposable input commit changed"
        elif change.path in _NONE:
            expected_severity, reason = "NONE", "per-run timing or identity metadata"
        if expected_severity is not None:
            _require(change.severity == expected_severity, "declared delta severity mismatch")
        item = change.model_dump(mode="json") | {"explanation": reason}
        explained.append(item)
        if expected_severity is None:
            extras.append(item)
    required = {field, f"files.{config_path}.sha256", "code.commit"}
    _require(required <= expected_paths, "required experimental or input delta missing")
    for audit in audits:
        for rule in (
            "consistency.env_vs_lock",
            "consistency.file_hashes",
            "consistency.lock_fresh",
            "consistency.model_identity",
            "consistency.generation_params",
            "consistency.custom_fields",
        ):
            findings = [item for item in audit.findings if item.rule_id == rule]
            if rule in (
                "consistency.env_vs_lock",
                "consistency.file_hashes",
                "consistency.lock_fresh",
                "consistency.model_identity",
            ):
                _require(
                    bool(findings) and all(item.status.value == "pass" for item in findings),
                    "required consistency result is not PASS",
                )
            _require(
                all(item.status.value in ("pass", "skipped") for item in findings),
                "pair consistency contradicts declared runtime",
            )
    return {
        "schema_version": 1,
        "status": "NEEDS_REVIEW" if extras else "PASS",
        "effective_bindings": effective,
        "target_gold_unchanged": True,
        "complete_changes": explained,
        "additional_changes_requiring_review": extras,
        "self_diff_empty": True,
        "full_findings_equal": documents[2]["findings"] == documents[3]["findings"],
        "audit_profiles_equal": documents[2]["profiles"] == documents[3]["profiles"],
        "audit_summaries": [item["summary"] for item in documents[2:]],
        "observed_delta_count": len(pair.changes),
    }
