"""Independent conservative path checks with explicit Python lexical context.

Arithmetic operators, the fixed [/INST] token, and f-string continuations
of a demonstrably relative path are syntax rather than machine paths.
"""
import ast
import io
import re
import tokenize

LOCAL_PATH = re.compile(
    r'(?i)\bfile:[/\\]{2}'
    r'|(?<![\w])[A-Za-z]:[\\/](?=[^\s\"\x27<>;,\)\]}>])'
    r'|(?<![\w:/\\*<.])/(?=[^\s\"\x27<>;,\)\]}>])[^\s\"\x27<>;,\)\]}>]*'
    r'|\\\\[A-Za-z0-9_.-]+\\+[A-Za-z0-9_.-]+'
)


def _simple_name(node):
    return isinstance(node, ast.Name) or isinstance(node, ast.Attribute) and _simple_name(node.value)


def _safe_fstring(token):
    try:
        value=ast.parse(token, mode='eval').body
    except SyntaxError:
        return False
    if not isinstance(value,ast.JoinedStr) or not value.values:
        return False
    first=value.values[0]
    if not isinstance(first,ast.Constant) or not isinstance(first.value,str):
        return False
    if not re.match(r'^[A-Za-z0-9_.-]+/',first.value):
        return False
    for part in value.values:
        if isinstance(part,ast.Constant):
            if not isinstance(part.value,str) or re.fullmatch(r'[A-Za-z0-9_./{}-]*',part.value) is None:
                return False
        elif isinstance(part,ast.FormattedValue):
            if not _simple_name(part.value) or part.format_spec is not None:
                return False
        else:
            return False
    return True


def _python_spans(text):
    try:
        ast.parse(text)
    except (SyntaxError,ValueError,RecursionError):
        return [],[]
    offsets=[0]
    for line in text.splitlines(keepends=True):
        offsets.append(offsets[-1]+len(line))
    def offset(position):
        row,column=position
        return offsets[row-1]+column
    operators=[]
    fstrings=[]
    try:
        for token in tokenize.generate_tokens(io.StringIO(text).readline):
            start,end=offset(token.start),offset(token.end)
            if token.type==tokenize.OP and token.string in {'/','//','/=','//='}:
                operators.append((start,end))
            elif token.type==tokenize.STRING and _safe_fstring(token.string):
                fstrings.append((start,end))
    except (tokenize.TokenError,IndentationError,IndexError):
        return [],[]
    return operators,fstrings


def unsafe_matches(text, *, python_source=False):
    operators,fstrings=_python_spans(text) if python_source else ([],[])
    result=[]
    for match in LOCAL_PATH.finditer(text):
        start,end=match.span()
        if match.group()=='/INST' and start>0 and text[start-1:end+1]=='[/INST]':
            continue
        if any(left==start and end<=right for left,right in operators):
            continue
        if any(left<=start<end<=right for left,right in fstrings):
            continue
        result.append(match)
    return result
