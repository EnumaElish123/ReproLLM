def digest(value):
    body = value if isinstance(value, bytes) else json.dumps(value, sort_keys=True, ensure_ascii=False, default=str).encode()
    return hashlib.sha256(body).hexdigest()


def normalize(value, schema, root_schema):
    if '$ref' in schema:
        schema = root_schema['$defs'][schema['$ref'].split('/')[-1]]
    if 'anyOf' in schema:
        if value is None:
            return None
        candidates = [s for s in schema['anyOf'] if s.get('type') != 'null']
        schema = candidates[0] if candidates else {}
        return normalize(value, schema, root_schema)
    if isinstance(value, dict):
        result = copy.deepcopy(value)
        # Exported JSON Schema omits factory defaults; MetricSpec.params is
        # explicitly Field(default_factory=dict) in the bound manifest schema.
        if schema.get('title') == 'MetricSpec':
            result.setdefault('params', {})
        for key, subschema in schema.get('properties', {}).items():
            if key not in result and 'default' in subschema:
                result[key] = copy.deepcopy(subschema['default'])
            if key in result:
                result[key] = normalize(result[key], subschema, root_schema)
        for pattern, subschema in schema.get('patternProperties', {}).items():
            for key in set(result) - set(schema.get('properties', {})):
                if re.search(pattern, key):
                    result[key] = normalize(result[key], subschema, root_schema)
        extra = schema.get('additionalProperties')
        if isinstance(extra, dict):
            for key in set(result)-set(schema.get('properties', {})):
                result[key] = normalize(result[key], extra, root_schema)
        return result
    if isinstance(value, list):
        return [normalize(v, schema.get('items', {}), root_schema) for v in value]
    if schema.get('type') == 'number' and isinstance(value, (int, float)) and not isinstance(value, bool):
        return float(value)
    return value


def plain(value):
    if isinstance(value, dict):
        if {'value', 'source', 'confidence'} <= set(value):
            return plain(value['value'])
        return {k:plain(v) for k,v in sorted(value.items()) if v is not None}
    if isinstance(value, list):
        return [plain(v) for v in value]
    return value


def project(value, prefix, source, flat):
    if isinstance(value, dict):
        if source == 'lock' and {'value', 'source', 'confidence'} <= set(value):
            flat[prefix] = {'value':value['value'], 'source':source, 'confidence':value['confidence'], 'detail':value['source']}
            return
        for key, child in sorted(value.items()):
            if source == 'lock' and key == 'observed_at':
                continue
            project(child, (prefix+'.'+key).lstrip('.'), source, flat)
    elif value is not None:
        flat[prefix] = {'value':plain(value), 'source':source, 'confidence':'observed' if source=='run' else 'declared', 'detail':None}


def rank(leaf):
    if leaf['source'] == 'run':
        return {'cli':9, 'config':8, 'env':7}.get((leaf.get('detail') or '').split(':')[0], 6)
    return {'lock':4, 'manifest':3}[leaf['source']]


def key(leaf):
    return (-rank(leaf), leaf.get('detail') or '', json.dumps(leaf['value'],sort_keys=True,ensure_ascii=False))


def snapshot_flat(manifest, lock, run):
    declared = {};locked = {};observed = {}
    for section in SECTIONS:
        project(manifest.get(section), section, 'manifest', declared)
        project(lock.get(section), section, 'lock', locked)
    for name in ['python','platform','packages']:
        project((lock.get('environment') or {}).get(name), 'environment.'+name, 'lock', locked)
    project((lock.get('environment') or {}).get('gpu'), 'hardware', 'lock', locked)
    for f in lock.get('files',[]):
        locked['files.'+f['path']+'.sha256']={'value':f['sha256'],'source':'lock','confidence':'declared','detail':None}
    for name in ['code','environment','command','run_id','started_at','ended_at','duration_seconds']:
        project(run.get(name), name, 'run', observed)
    hardware = copy.deepcopy(run.get('hardware') or {})
    hardware['gpus'] = {str(g['index']):g for g in hardware.get('gpus', [])}
    project(hardware, 'hardware', 'run', observed)
    for f in run.get('files',[]):
        observed['files.'+f['path']+'.sha256']={'value':f['sha256'],'source':'run','confidence':'observed','detail':None}
    for field, items in sorted(run.get('bindings_observed',{}).items()):
        leaves=[]
        for item in items:
            src=item['source'];where=(src.get('path','')+':') if src.get('path') else ''
            leaves.append({'value':item['value'],'source':'run','confidence':'observed','detail':src['type']+':'+where+src['key']})
        if not leaves:
            continue
        if field in observed:
            leaves.append(observed[field])
        cli=[n for n,l in enumerate(leaves) if rank(l)==9]
        observed[field]=leaves[cli[-1]] if cli else min(leaves,key=key)
    fields=set(declared)|set(locked)|set(observed)
    return {f:min([d[f] for d in [declared,locked,observed] if f in d],key=key) for f in sorted(fields)}


def section(document, title):
    lines=document.splitlines();prefixes=['## '+title,'### '+title,'#### '+title]
    hits=[n for n,l in enumerate(lines) if l in prefixes]
    if not hits:
        return ''
    assert len(hits)==1, 'Repeated section '+title
    n=hits[0];level=len(lines[n].split(' ')[0]);end=len(lines)
    for i in range(n+1,len(lines)):
        if re.match(r'^#{1,4} ',lines[i]) and len(lines[i].split(' ')[0])<=level:
            end=i;break
    return '\n'.join(lines[n+1:end])


def literal(value):
    return html.unescape(value).replace('<br>','\n').replace('\\|','|').replace('\\\\','\\')


def table(text):
    result=[]
    for line in text.splitlines():
        if not line.startswith('| ') or line.startswith('|---'):
            continue
        result.append([literal(x.strip()) for x in re.split(r'(?<!\\)\|',line)[1:-1]])
    return result


def first_table(text):
    lines=text.splitlines();start=next((i for i,l in enumerate(lines) if l.startswith('| ')),None)
    if start is None:return []
    end=start
    while end<len(lines) and lines[end].startswith('|'):end+=1
    return table('\n'.join(lines[start:end]))


def bullets(text):
    result={}
    for line in text.splitlines():
        match=re.fullmatch(r'- (`+)(.*?)\1: (.*)',line)
        if match:
            assert match[2] not in result, 'Repeated bullet key'
            result[match[2]]=literal(match[3])
    return result


def short(value):
    text=str(value or '')
    if re.fullmatch(r'sha256:[a-fA-F0-9]{64}',text):return text[:19]
    if re.fullmatch(r'[a-fA-F0-9]{32,}',text):return text[:12]
    return text


def evidence(leaf):
    return {'run':'observed (run)','manifest':'declared (manifest)'}.get(leaf['source'],'locked ('+leaf['confidence']+')')


def revision(flat, field, detailed=True):
    leaf=flat.get(field)
    if leaf is None:return 'not locked'
    if leaf['source']=='run':return short(leaf['value'])+' (observed)'
    if leaf['source']=='lock':
        if leaf['confidence']=='exact':return short(leaf['value'])
        return 'not locked ('+leaf['confidence']+')' if detailed else 'not locked'
    return 'not locked'
