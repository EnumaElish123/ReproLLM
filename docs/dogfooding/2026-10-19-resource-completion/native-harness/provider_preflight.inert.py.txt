"""Bounded, read-only provider checks; never save response bodies or credentials."""
import hashlib
import json
import os
from pathlib import Path
import time

import httpx

OUT = Path('<TASK>/providers')
OUT.mkdir(mode=0o700, exist_ok=True)
results = {'schema_version': 1, 'paid_requests': 0, 'checks': []}

def check(label, url, token, bounded_file=False):
    row = {'label': label, 'credential_present': bool(token)}
    if not token:
        row['status'] = 'BLOCKED_MISSING_CREDENTIAL'
        results['checks'].append(row)
        return
    started = time.monotonic()
    try:
        with httpx.Client(timeout=20, follow_redirects=False, trust_env=False) as client:
            with client.stream('GET', url, headers={'Authorization': 'Bearer ' + token}) as response:
                row['http_status'] = response.status_code
                if bounded_file:
                    data = bytearray()
                    if response.status_code == 200:
                        for chunk in response.iter_bytes():
                            if len(data) + len(chunk) > 2 * 1024**2:
                                raise ValueError('size bound')
                            data.extend(chunk)
                        row['size_bytes'] = len(data)
                        row['sha256'] = hashlib.sha256(data).hexdigest()
                        row['valid_json'] = isinstance(json.loads(data), dict)
                    row['status'] = 'ACCESSIBLE' if response.status_code == 200 else 'ACCESS_DENIED' if response.status_code in (401,403) else 'PROVIDER_UNAVAILABLE'
                else:
                    data = response.read()
                    if len(data) > 1024**2:
                        raise ValueError('size bound')
                    body = json.loads(data)
                    if label == 'deepseek_models' and response.status_code == 200:
                        row['selected_model_available'] = any(m.get('id') == 'deepseek-flash' for m in body.get('data', []))
                    row['status'] = 'AVAILABLE' if response.status_code == 200 else 'ACCESS_DENIED'
    except Exception as error:
        row['status'] = 'CHECK_FAILED'
        row['error_type'] = type(error).__name__
    row['elapsed_seconds'] = round(time.monotonic()-started, 3)
    results['checks'].append(row)

hf = os.environ.get('HF_TOKEN') or os.environ.get('HUGGING_FACE_HUB_TOKEN')
deepseek = os.environ.get('DEEPSEEK_API_KEY')
check('huggingface_auth', 'https://huggingface.co/api/whoami-v2', hf)
for filename in ('config.json', 'tokenizer_config.json'):
    check('llama2_' + filename, 'https://huggingface.co/meta-llama/Llama-2-7b-hf/resolve/01c7f73d771dfac7d292323805ebc428287df4f9/' + filename, hf, True)
check('deepseek_models', 'https://api.deepseek.com/models', deepseek)
raw = (json.dumps(results, indent=2, sort_keys=True) + '\n').encode()
for token in (hf, deepseek):
    assert not token or token.encode() not in raw
target=OUT/'preflight.json'
with target.open('xb') as stream:
    stream.write(raw)
print(raw.decode())
