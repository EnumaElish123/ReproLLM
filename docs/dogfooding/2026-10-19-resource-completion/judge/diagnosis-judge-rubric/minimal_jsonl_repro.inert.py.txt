"""CPU-only rubric JSONL fidelity reproducer; no secret, network or file writes."""
import json
from pathlib import Path
import sys
from reprollm.run.privacy import RunPrivacy

original={'prompt_template':'### Assistant A:\n{answer}'}
source=json.dumps(original)+'\n'
privacy=RunPrivacy(Path.cwd(),hostname='synthetic-host',username='synthetic-user')
saved,count=privacy.text(source)
actual=json.loads(saved)
if actual!=original:
 print(json.dumps({'schema_version':1,'status':'REPRODUCED','input_jsonl':source,'expected_decoded_value':original,'actual_decoded_value':actual,'redactions':count,'real_machine_path_or_secret':False},sort_keys=True))
 sys.exit(1)
print(json.dumps({'schema_version':1,'status':'PASS'}))
