def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def strict_json(raw: bytes) -> Any:
    def unique(pairs):
        result = {}
        for key, value in pairs:
            if key in result:
                raise ValueError("duplicate JSON key")
            result[key] = value
        return result
    def finite(value):
        raise ValueError("non-finite JSON constant")
    def finite_float(value):
        result = float(value)
        if not math.isfinite(result):
            raise ValueError("non-finite JSON number")
        return result
    return json.loads(raw.decode("utf-8", "strict"), object_pairs_hook=unique,
                      parse_constant=finite, parse_float=finite_float)


def write_json(path: Path, value: dict) -> None:
    assert value["schema_version"] == 1
    path.write_text(json.dumps(value, indent=2, sort_keys=True, ensure_ascii=False) + "\n", encoding="utf-8")


def source_inventory(candidate: Path, files: dict[str, str]) -> None:
    for relative, expected in files.items():
        path = candidate / relative
        assert not Path(relative).is_absolute() and ".." not in Path(relative).parts
        assert path.is_file() and sha(path.read_bytes()) == expected, "Final QA input bytes changed"


def cache_inventory(bundle: Path, preparation: Path) -> dict:
    index_raw = (bundle / "index.json").read_bytes()
    assert sha(index_raw) == INDEX_SHA, "Independent oracle index mismatch"
    index = strict_json(index_raw)
    prepared = strict_json(preparation.read_bytes())
    assert prepared["schema_version"] == 1 and prepared["all_five_valid_clean"]
    assert prepared["oracle_bundle_index_sha256"] == INDEX_SHA
    expected_names = set(PINS)
    assert len(prepared["targets"])==5 and {row["target"] for row in prepared["targets"]} == expected_names
    for target in prepared["targets"]:
        name = target["target"]
        assert target["expected_sha"] == PINS[name][0]
        assert len(target["expected_cache"])==2
        assert {file["relative_file"] for file in target["expected_cache"]}=={f"expected-audit/{name}.json",f"expected-facts/{name}.json"}
        for file in target["expected_cache"]:
            rel = file["relative_file"]
            raw = (bundle / rel).read_bytes()
            assert sha(raw) == file["sha256"] == index["files_sha256"][rel]
            assert strict_json(raw)["schema_version"] == 1
    provenance_rel = "gate-a/oracle-and-environment-provenance.json"
    assert sha((bundle / provenance_rel).read_bytes()) == index["files_sha256"][provenance_rel]
    return {"schema_version":1,"index_sha256":INDEX_SHA,"preparation_sha256":sha(preparation.read_bytes()),
            "expected_files":{file["relative_file"]:file["sha256"] for target in prepared["targets"] for file in target["expected_cache"]},
            "provenance_sha256":sha((bundle/provenance_rel).read_bytes())}

    def portable(self,value):
        if isinstance(value,str):
            for prefix,token in self.mapping:value=value.replace(prefix,token)
            return value
        if isinstance(value,list):return [self.portable(x) for x in value]
        if isinstance(value,dict):return {key:self.portable(x) for key,x in value.items()}
        return value

    def flush(self):
        write_json(self.out/'commands.json',self.portable({"schema_version":1,"commands":self.records}))
        write_json(self.out/'checks.json',self.portable({"schema_version":1,"checks":self.checks}))

    def check(self,name,label,ok,**proof):
        self.checks.append({"target":name,"label":label,"passed":bool(ok),**proof})
        self.flush()
        if not ok:print("FAIL "+name+" "+label,flush=True)

    def command(self,name,label,argv,cwd,kind="preparation",expected_exit=0):
        source_inventory(self.args.candidate,self.auth['candidate_files_sha256'])
        stem=f"{len(self.records)+1:03d}-{name}-{label}"
        started=time.monotonic()
        started_epoch=time.time()
        timed_out=False
        try:
            result=subprocess.run(argv,cwd=cwd,env=self.env,stdout=subprocess.PIPE,stderr=subprocess.PIPE,
                                  timeout=300,check=False)
            code,stdout,stderr=result.returncode,result.stdout,result.stderr
        except subprocess.TimeoutExpired as exc:
            timed_out=True;code=None;stdout=exc.stdout or b"";stderr=exc.stderr or b""
        elapsed=round(time.monotonic()-started,6)
        unicode_valid=True
        for stream,raw in (("stdout",stdout),("stderr",stderr)):
            try:raw.decode('utf-8','strict')
            except UnicodeError:unicode_valid=False
            (self.out/'raw'/f'{stem}.{stream}').write_bytes(raw)
        record={"target":name,"label":label,"kind":kind,"argv":list(map(str,argv)),"cwd":str(cwd),
                "exit_code":code,"expected_exit":expected_exit,"elapsed_seconds":elapsed,"started_epoch":started_epoch,"ended_epoch":time.time(),"timed_out":timed_out,
                "full_utf8":unicode_valid,"stdout":f'raw/{stem}.stdout',"stderr":f'raw/{stem}.stderr',
                "stdout_sha256":sha(stdout),"stderr_sha256":sha(stderr),"stdout_bytes":len(stdout),"stderr_bytes":len(stderr)}
        self.records.append(record);self.flush()
        self.check(name,label+"-exit-and-utf8",code==expected_exit and unicode_valid and not timed_out)
        print(f"{kind} {name} {label} exit={code} seconds={elapsed}",flush=True)
        return {"stdout":stdout,"stderr":stderr,"exit_code":code,"record":record}

    def cli(self,name,label,argv,cwd,expected_exit=0):
        return self.command(name,label,[str(self.args.python),str(self.launcher),*map(str,argv)],cwd,
                            kind="fresh-product",expected_exit=expected_exit)

    def pin(self,name,phase):
        values=[]
        for label,argv in [("head",['rev-parse','HEAD']),("origin",['remote','get-url','origin']),
                           ("status",['status','--porcelain=v1','--untracked-files=all'])]:
            r=self.command(name,f"pin-{phase}-{label}",['git','-C',str(self.args.portfolio/name),*argv],self.args.candidate)
            values.append(r['stdout'].decode('utf-8','strict').strip())
        commit,upstream=PINS[name]
        valid=values[0]==commit and values[2]=='' and values[1].lower() in {('https://github.com/'+upstream+'.git').lower(), ('https://github.com/'+upstream).lower(), ('git@github.com:'+upstream+'.git').lower(), ('ssh://git@github.com/'+upstream+'.git').lower()}
        self.check(name,'pin-'+phase+'-identity',valid,commit=values[0],origin_sha256=sha(values[1].encode()),clean=values[2]=='')
        return {"target":name,"commit":values[0],"origin_sha256":sha(values[1].encode()),"clean":values[2]=='',"valid":valid}

    def compare_base(self,name,label,result):
        try:
            actual=strict_json(result['stdout'])
            valid=isinstance(actual,dict) and actual.get('schema_version')==1
            if label=='L0':
                valid=valid and actual.get('reprollm_version')==self.auth['version'] and bool(re.fullmatch(r'\d{4}-\d\d-\d\dT\d\d:\d\d:\d\dZ',actual.get('generated_at','')))
            expected_stderr=b''
            if label=='L0' and name=='lm-evaluation-harness':
                expected_stderr=b'scan: Python analysis incomplete: parsed 500 of 816 candidate files; selected 500 (limit 500), omitted 316. Missing signals may be in unanalysed files.\n'
            self.check(name,label+'-single-complete-json',valid and result['stderr']==expected_stderr,
                       stderr_expected_sha256=sha(expected_stderr))
            expected=self.args.oracle_bundle/('expected-audit' if label=='L0' else 'expected-facts')/(name+'.json')
            actual_file=self.out/result['record']['stdout']
            argv=[str(self.args.python),str(self.args.candidate/'scripts/compare_gold.py'),str(expected),str(actual_file)]
            if label=='L0':argv.append('--audit-metadata')
            compared=self.command(name,label+'-complete-comparison',argv,self.args.candidate,kind='comparison')
            value=strict_json(compared['stdout'])
            self.check(name,label+'-complete-fields-equal',compared['exit_code']==0 and compared['stderr']==b'' and value.get('status')=='PASS' and value.get('differences')==[],
                       expected_sha256=sha(expected.read_bytes()),actual_sha256=sha(result['stdout']),comparison_ref=compared['record']['stdout'])
        except (ValueError,UnicodeError,OSError,AssertionError,KeyError) as exc:
            self.check(name,label+'-complete-comparison-failed',False,error_type=type(exc).__name__)

    def qualify(self,name,label,result,fixture):
        try:
            proof=self.oracle.check_leaf(name,label,result,fixture,self.auth)
            assert isinstance(proof,dict) and proof.get('schema_version')==1
            assert proof.get('complete_output_checked') is True
            assert isinstance(proof.get('checks'),list) and proof['checks']
            assert proof.get('oracle_source') and proof.get('fields_or_bytes_covered')
            write_json(self.out/'comparison'/f'{name}-{label}.json',self.portable(proof))
            self.check(name,label+'-independent-full-oracle',all(x.get('passed') is True for x in proof['checks']),
                       proof_ref=f'comparison/{name}-{label}.json',proof_sha256=sha((self.out/'comparison'/f'{name}-{label}.json').read_bytes()))
        except Exception as exc:
            self.check(name,label+'-oracle-failure',False,error_type=type(exc).__name__)

    def capture_run(self,name,which,result,fixture):
        root=fixture['root']
        existing=fixture.setdefault('seen_ids',set())
        paths=list(root.glob('.reprollm/runs/*/run.json'))
        new=[p for p in paths if p.parent.name not in existing]
        assert len(new)==1,'Each run must create exactly one new record'
        path=new[0];raw=path.read_bytes();record=strict_json(raw)
        assert record['run_id']==path.parent.name and record['schema_version']==2
        if which=='B':
            assert record['started_at']>fixture['run_records']['A']['started_at'], 'B must be unambiguously newer than A'
        existing.add(record['run_id'])
        fixture['run_records'][which]=record
        fixture['artifacts']['run-'+which]=path
        witness_raw=(root/'r04-consumed.json').read_bytes()
        witness=strict_json(witness_raw)
        assert witness=={'schema_version':1,'consumed_temperature':0.2 if which=='A' else 0.1,'config_before':{'temperature':0.1},'torch_metadata':'0.0.123','rewritten':which=='B'}
        fixture['witnesses'][which]=witness
        write_json(self.out/'artifacts'/f'{name}-witness-{which}.json',witness)
        # Preserve all nested version-2 fields. No version stripping/model dump.
        envelope={'schema_version':1,'run':record}
        write_json(self.out/'artifacts'/f'{name}-run-{which}-envelope.json',envelope)
        fixture['artifacts']['envelope-'+which]=self.out/'artifacts'/f'{name}-run-{which}-envelope.json'
        fixture['record_raw_sha256_'+which]=sha(raw)
        expected_bytes=CONFIG_AFTER if which=='B' else CONFIG_BEFORE
        assert (root/'r04-config.json').read_bytes()==expected_bytes
        metadata=self.command(name,'recorder-host-control-after-'+which,[str(self.args.python),str(self.control),str(self.args.candidate)],root)
        assert metadata['exit_code']==0
        fixture['metadata']=strict_json(metadata['stdout'])
        self.qualify(name,'run-'+which,result,fixture)

    def pin_matrix(self,name):

        r=self.cli(name,'L0',['audit','.','--level','0','--format','json','--fail-on','never'],target)

        r=self.command(name,'facts',[str(self.args.python),str(self.facts),str(self.args.candidate)],target,kind='fresh-product')
        self.compare_base(name,'facts',r)
        f=self.clone(name);root=f['root']
        r=self.cli(name,'lock-offline',['lock','.','--offline'],root)
        self.qualify(name,'lock-offline',r,f)
        f['input_before']['reprollm.lock']=(root/'reprollm.lock').read_bytes()
        base=[str(f['child_python']),'r04-child.py','--config','r04-config.json']
        self.dry_run(name,root,base)
        r=self.cli(name,'run-A',['run','--name','s4a-evidence-A','--capture-output','--',*base,'--temperature','0.1','--temperature','0.2'],root)
        self.capture_run(name,'A',r,f)
        r=self.cli(name,'L2-A',['audit','.','--format','json','--fail-on','never'],root)
        f['artifacts']['audit-A']=self.out/r['record']['stdout'];self.qualify(name,'L2-A',r,f)
        # Run ordering uses second-precision start times plus ID. Keep A/B
        # strictly ordered without rewriting timestamps or random identifiers.
        previous=datetime.fromisoformat(f['run_records']['A']['started_at'].replace('Z','+00:00')).timestamp()
        wait_start=time.monotonic()
        while time.time()<int(previous)+1:
            assert time.monotonic()-wait_start<2, 'UTC clock could not establish A/B ordering'
            time.sleep(0.05)
        f['ordering_control']={'method':'wait for UTC second after A.started_at',
                               'elapsed_seconds':round(time.monotonic()-wait_start,6)}
        r=self.cli(name,'run-B',['run','--name','s4a-evidence-B','--capture-output','--',*base,'--rewrite'],root)
        self.capture_run(name,'B',r,f)
        r=self.cli(name,'L2-B',['audit','.','--format','json','--fail-on','never'],root)
        f['artifacts']['audit-B']=self.out/r['record']['stdout'];self.qualify(name,'L2-B',r,f)
        aid=f['run_records']['A']['run_id'];bid=f['run_records']['B']['run_id']
        for label,args,exit_code in [('pair-json',['diff',aid,bid,'--format','json','--fail-on','HIGH'],1),
                                     ('self-json',['diff',str(f['artifacts']['run-B']),str(f['artifacts']['run-B']),'--format','json','--fail-on','HIGH'],0),
                                     ('pair-text',['--no-color','diff',aid,bid,'--fail-on','HIGH'],1)]:
            r=self.cli(name,label,args,root,exit_code);self.qualify(name,label,r,f)
        for which,venue in [('A','default'),('B','default')]:
            label=f'export-{which}-{venue}';path=self.out/'artifacts'/f'{name}-{which}-{venue}.md'
            r=self.cli(name,label,['export','.','--run',f['run_records'][which]['run_id'],'--template',venue,'--output',str(path)],root)
            f['artifacts'][label]=path;self.qualify(name,label,r,f)

    def execute(self):
        before={name:self.pin(name,'before') for name in PINS}
        write_json(self.out/'pins-before.json',{'schema_version':1,'pins':list(before.values())})
        for name in PINS:
            try:
                if not before[name]['valid']:continue
                self.pin_matrix(name)
            except Exception as exc:
                self.check(name,'pin-matrix-incomplete',False,error_type=type(exc).__name__)
        after={name:self.pin(name,'after') for name in PINS}
        write_json(self.out/'pins-after.json',{'schema_version':1,'pins':list(after.values())})
        for name in PINS:self.check(name,'pin-before-after-equal',before[name]==after[name])
        source_inventory(self.args.candidate,self.auth['candidate_files_sha256'])
        leaves=[r for r in self.records if r['kind']=='fresh-product']
        coverage={name:sorted(r['label'] for r in leaves if r['target']==name) for name in PINS}
        complete=all(labels==sorted(LABELS) for labels in coverage.values())
        passed=complete and all(x['passed'] for x in self.checks)
        summary={'schema_version':1,'status':'PASS' if passed else 'FAIL','all_passed':passed,
                 'fresh_product_leaf_commands':len(leaves),'required_core_product_leaf_commands':75,
                 'leaf_coverage':coverage,'checks':len(self.checks),'failed_checks':sum(not c['passed'] for c in self.checks),
                 'affected_consumer_commands':sum(r['kind']=='affected-consumer' for r in self.records),
                 'affected_consumer_seconds':round(sum(r['elapsed_seconds'] for r in self.records if r['kind']=='affected-consumer'),6),
                 'product_seconds':round(sum(r['elapsed_seconds'] for r in leaves),6),'wall_seconds':round(time.monotonic()-self.started,6),
                 'pins_valid_clean_before_after':all(row['valid'] for row in [*before.values(),*after.values()]),
                 'candidate_inputs_unchanged':True,'runner_sha256':sha(Path(__file__).read_bytes()),
                 'authorization_sha256':sha(self.args.authorization.read_bytes()),'quality_receipt_sha256':self.auth['quality_receipt_sha256'],
                 'extension_oracle_sha256':self.auth['extension_oracle_sha256'],'resource_gpu_api_commands':0,
                 'old_product_runs_reused':0,'old_gold_or_pins_updated':False,'bridge_probes':'SEPARATE_NOT_COUNTED',
                 'raw_payload_retention':'One full stdout/stderr copy per command; complete run2 in one schema1 envelope per run.'}
        write_json(self.out/'summary.json',summary)
        return 0 if passed else 1


def execute(args) -> int:
    auth=strict_json(args.authorization.read_bytes())
    assert auth['schema_version']==1 and auth['execution_authorized'] is True and auth['final_quality_pass'] is True
    assert auth['runner_sha256']==sha(Path(__file__).read_bytes())
    assert auth['quality_receipt_sha256']==sha(args.quality_receipt.read_bytes())
    assert auth['quality_inputs_sha256']==sha(args.quality_inputs.read_bytes())
    assert auth['extension_oracle_sha256']==sha(args.extension_oracle.read_bytes())
    assert auth['harness_companion_files_sha256']=={filename:sha((HERE/filename).read_bytes()) for filename in COMPANION_FILES}
    assert auth['oracle_review_sha256']==sha(args.oracle_review.read_bytes())
    oracle_review=strict_json(args.oracle_review.read_bytes())
    assert oracle_review['schema_version']==1 and oracle_review['complete_75_leaf_oracle_reviewed'] is True
    assert oracle_review['extension_oracle_sha256']==sha(args.extension_oracle.read_bytes())
    assert set(oracle_review['pins'])==set(PINS)
    assert all(oracle_review['pins'][name]==PINS[name][0] for name in PINS)
    assert sorted(oracle_review['labels'])==sorted(LABELS) and len(oracle_review['labels'])==15
    assert oracle_review['harness_companion_files_sha256']==auth['harness_companion_files_sha256']
    quality=strict_json(args.quality_receipt.read_bytes())
    assert quality['schema_version']==1 and (quality.get('all_passed') is True or quality.get('status')=='PASS')
    quality_inputs=strict_json(args.quality_inputs.read_bytes())
    assert quality_inputs['schema_version']==1
    declared=quality_inputs.get('files_sha256',quality_inputs.get('files'))
    assert isinstance(declared,dict) and declared
    quality_files={path:(item['sha256'] if isinstance(item,dict) else item) for path,item in declared.items()}
    assert quality_files==auth['candidate_files_sha256'], 'Authorized source set differs from actual final QA inputs'
    inventory_hash=quality_inputs.get('inventory_sha256')
    assert isinstance(inventory_hash,str) and len(inventory_hash)==64
    recomputed_inventory_hash=sha(json.dumps(declared,sort_keys=True,separators=(',', ':')).encode())
    assert inventory_hash==recomputed_inventory_hash, 'QA inventory content does not match its canonical digest'
    assert inventory_hash==auth['quality_input_inventory_sha256']
    quality_inventory=quality.get('candidate_input_inventory_sha256',quality.get('input_inventory_sha256',quality.get('inventory_sha256')))
    assert quality_inventory==inventory_hash, 'Quality PASS receipt belongs to different input bytes'
    assert auth['version']=='0.7.0', 'Final R04-B minor writer source required'
    assert auth['candidate_files_sha256'] and any(p.startswith('src/') for p in auth['candidate_files_sha256'])
    assert any(p.startswith('tests/') for p in auth['candidate_files_sha256'])
    assert any(p.startswith('schemas/') for p in auth['candidate_files_sha256'])
    source_inventory(args.candidate,auth['candidate_files_sha256'])
    assert sha(args.git_shim.read_bytes())==SHIM_SHA
    assert sha(args.facts_source.read_bytes())=="3fc111446cc13bf77dcc298118a8cab853ad4daaf67c3c304fb052194e5d3d60"
    cache_inventory(args.oracle_bundle,args.preparation)
    # The separately audited oracle may not import product modules/functions.
    tree=ast.parse(args.extension_oracle.read_text(encoding='utf-8'))
    for node in ast.walk(tree):
        if isinstance(node,ast.Import):assert all(not n.name.startswith('reprollm') for n in node.names)
        if isinstance(node,ast.ImportFrom):assert not (node.module or '').startswith('reprollm')
    assert not args.output.exists(), 'Refuse previous execution overwrite'
    args.output.mkdir(parents=True)
    for sub in ['raw','harness','comparison','artifacts','clones','child-envs']:(args.output/sub).mkdir()
    return Gate(args,auth).execute()
