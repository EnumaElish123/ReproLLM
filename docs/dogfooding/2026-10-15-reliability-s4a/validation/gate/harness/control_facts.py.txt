"""Independent recorder/host metadata; stdlib only, no product imports."""
from __future__ import annotations
import ast,csv,hashlib,importlib.metadata,io,json,os,platform,re,socket,subprocess,sys,time
from pathlib import Path
sha=lambda data:'sha256:'+hashlib.sha256(data).hexdigest()
candidate=Path(sys.argv[1]);tree=ast.parse((candidate/'src/reprollm/core/envinfo.py').read_bytes())
literals={}
for node in tree.body:
    name=node.target.id if isinstance(node,ast.AnnAssign) and isinstance(node.target,ast.Name) else None
    if name in {'LLM_CRITICAL_PACKAGES','_DISTRIBUTION_NAMES'}:literals[name]=ast.literal_eval(node.value)
packages={}
for name in literals['LLM_CRITICAL_PACKAGES']:
    try:packages[name]=importlib.metadata.version(literals['_DISTRIBUTION_NAMES'].get(name,name))
    except importlib.metadata.PackageNotFoundError:pass
calls=[]
def command(argv):
    started=time.monotonic()
    try:r=subprocess.run(argv,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=5)
    except (OSError,subprocess.TimeoutExpired):return None
    calls.append({'argv':argv,'exit':r.returncode,'elapsed_seconds':round(time.monotonic()-started,6),
                  'stdout_sha256':sha(r.stdout),'stderr_sha256':sha(r.stderr)})
    return r.stdout.decode('utf-8','strict') if r.returncode==0 else None
query=command(['nvidia-smi','--query-gpu=index,name,memory.total,uuid','--format=csv,noheader,nounits'])
drivers=command(['nvidia-smi','--query-gpu=driver_version','--format=csv,noheader'])
header=command(['nvidia-smi'])
hardware={'cpu_count':os.cpu_count() or 0,'gpus':[],'driver':None,'cuda_driver_max':None,'source':'unavailable'}
lock_gpu={'driver':None,'cuda_driver_max':None,'source':'unavailable'}
if drivers is not None:
    d=sorted({line.strip() for line in drivers.splitlines() if line.strip()})
    match=re.search(r'CUDA Version:\s*([\d.]+)',header or '')
    lock_gpu={'driver':d[0] if d else None,'cuda_driver_max':match.group(1) if match else None,'source':'nvidia_smi'}
    try:
        rows=list(csv.reader(io.StringIO(query or ''),skipinitialspace=True,strict=True))
        assert rows and all(len(row)==4 for row in rows) and len(d)==1 and re.fullmatch(r'\d+(?:\.\d+)+',d[0]) and match
        gpus=[{'index':int(row[0].strip()),'name':row[1].strip(),'memory_mib':int(row[2].strip()),'uuid_sha256':sha(row[3].strip().encode())} for row in rows]
        assert len({row['index'] for row in gpus})==len(gpus)
        hardware.update(gpus=sorted(gpus,key=lambda row:row['index']),driver=d[0],cuda_driver_max=match.group(1),source='nvidia_smi')
    except (AssertionError,ValueError,csv.Error):pass
expected_env={key:value for key,value in os.environ.items() if key.startswith(('OMP_','HF_HUB_OFFLINE','TRANSFORMERS_','PYTHON','LANG','LC_','TZ','PATH'))}
for key in ('PATH','PYTHONPATH'):
    if key in expected_env:expected_env[key]='<REDACTED:path>'
value={'schema_version':1,'python':platform.python_version(),'platform':'linux' if sys.platform.startswith('linux') else 'darwin' if sys.platform=='darwin' else 'windows',
       'os':platform.platform(),'hostname_sha256':sha(socket.gethostname().encode()),'packages':packages,
       'recorded_env':dict(sorted(expected_env.items())),'hardware':hardware,'lock_gpu':lock_gpu,
       'metadata_commands':calls,'product_imports':0,'heavy_imports':0,'gpu_compute':0,'network':0}
print(json.dumps(value,sort_keys=True,ensure_ascii=False))
