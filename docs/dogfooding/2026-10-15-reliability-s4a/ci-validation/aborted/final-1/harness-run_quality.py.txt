def digest(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def git(*args: str) -> bytes:
    return subprocess.run(
        ["/usr/bin/git", "-C", str(ROOT), *args],
        check=True,
        capture_output=True,
    ).stdout


def input_inventory(commit: str) -> dict[str, Any]:
    files: dict[str, dict[str, str]] = {}
    total = 0
    for entry in git("ls-files", "--stage", "-z").split(b"\0"):
        if not entry:
            continue
        metadata, name_raw = entry.split(b"\t", 1)
        mode, blob, stage = metadata.decode().split()
        assert stage == "0" and mode in {"100644", "100755", "120000"}
        name = name_raw.decode("utf-8", errors="strict")
        path = ROOT / name
        raw = (
            os.readlink(path).encode() if stat.S_ISLNK(path.lstat().st_mode) else path.read_bytes()
        )
        actual_blob = hashlib.sha1(b"blob " + str(len(raw)).encode() + b"\0" + raw).hexdigest()
        assert actual_blob == blob, "tracked bytes differ from committed index"
        files[name] = {"sha256": digest(raw), "git_blob_sha1": blob, "mode": mode}
        total += len(raw)
    assert files, "empty candidate inventory"
    canonical = json.dumps(files, sort_keys=True, separators=(",", ":")).encode()
    return {
        "schema_version": 1,
        "commit": commit,
        "tracked_file_count": len(files),
        "tracked_bytes_total": total,
        "files": files,
        "inventory_sha256": digest(canonical),
        "all_git_blob_bytes_verified": True,
    }


def portable(value: Any, mappings: list[tuple[str, str]], identities: list[str]) -> Any:
    if isinstance(value, str):
        for prefix, replacement in sorted(mappings, key=lambda item: -len(item[0])):
            value = value.replace(prefix, replacement)
        for identity in identities:
            value = re.sub(r"(?<!\w)" + re.escape(identity) + r"(?!\w)", "<LOCAL_IDENTITY>", value)
        return value
    if isinstance(value, list):
        return [portable(item, mappings, identities) for item in value]
    if isinstance(value, dict):
        return {
            portable(key, mappings, identities): portable(item, mappings, identities)
            for key, item in value.items()
        }
    return value

    examples = [ROOT / "examples" / name for name in (
        "hf_vllm_eval", "openai_judge_eval", "privacy_custom_params"

    assert all(path.is_dir() for path in examples)
    for example in examples:
        relative = example.relative_to(ROOT).as_posix()
        commands.extend(
            [

                    "example-" + example.name + "-audit",

        ("20-self-audit", [*cli, "audit", ".", "--format", "github", "--fail-on", "never"])

def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--expected-commit", required=True)
    parser.add_argument("--output", type=Path, default=DEFAULT_OUT)
    args = parser.parse_args()
    commit = args.expected_commit
    out = args.output.resolve()
    assert re.fullmatch(r"[0-9a-f]{40}", commit), "explicit full candidate commit required"
    assert out.is_relative_to(SCRATCH) and not out.is_relative_to(ROOT)
    assert git("rev-parse", "HEAD").decode().strip() == commit
    assert git("status", "--porcelain=v1", "--untracked-files=all") == b""
    assert digest(SHIM.read_bytes()) == EXPECTED_SHIM
    uv = shutil.which("uv")
    assert uv is not None, "uv executable unavailable"
    assert not any(
        (out / name).exists() for name in ("commands.json", "summary.json", "raw", "logs")
    ), "never overwrite an earlier attempt"
    out.mkdir(parents=True, exist_ok=True)
    (out / "raw").mkdir()
    (out / "logs").mkdir()
    mappings = [
        (str(out), "<QUALITY>"),
        (str(ROOT), "<CANDIDATE>"),
        (str(ENVIRONMENT), "<CANDIDATE_ENV>"),
        (str(SHIM.parent), "<GIT_SHIM>"),
        (uv, "<UV>"),
        (str(Path.home() / ".cache/uv"), "<UV_CACHE>"),
        (str(Path.home()), "<HOME>"),
        ("<SOURCE_REPOSITORY>", "<SOURCE_REPOSITORY>"),
    ]
    identities = [name for name in (getpass.getuser(), socket.gethostname()) if name]

    def safe(value: Any) -> Any:
        return portable(value, mappings, identities)

    def write(name: str, value: dict[str, Any]) -> None:
        assert value["schema_version"] == 1
        (out / name).write_text(json.dumps(safe(value), indent=2, sort_keys=True) + "\n")

    allowed_env = (
        "PATH",
        "HOME",
        "LANG",
        "LC_ALL",
        "TZ",
        "USER",
        "LOGNAME",
        "SystemRoot",
        "WINDIR",
        "PATHEXT",
    )
    env = {name: os.environ[name] for name in allowed_env if name in os.environ}
    env.update(
        {
            "PATH": str(SHIM.parent)
            + os.pathsep
            + str(PYTHON.parent)
            + os.pathsep
            + env.get("PATH", ""),
            "PYTHONPATH": str(ROOT / "src") + os.pathsep + str(ROOT),
            "PYTHONDONTWRITEBYTECODE": "1",
            "UV_PROJECT_ENVIRONMENT": str(ENVIRONMENT),
            "UV_NO_SYNC": "1",
            "COVERAGE_FILE": str(out / "coverage.data"),
            "PYTHONIOENCODING": "utf-8",
            "PYTHONUTF8": "1",
            "CUDA_VISIBLE_DEVICES": "",
            "HF_HUB_OFFLINE": "1",
            "HF_DATASETS_OFFLINE": "1",
        }
    )
    started_all = time.monotonic()
    harness_sha = digest(Path(__file__).read_bytes())
    before = input_inventory(commit)
    write("inputs-before.json", before)
    prior_path = SCRATCH / "quality/final-1/inputs-before.json"
    prior = json.loads(prior_path.read_bytes())
    assert prior["commit"] == "b86c014cf9f15f1efddb6902979dcc6ddf79be9f"
    def retained_inputs(inventory: dict[str, Any]) -> dict[str, Any]:
        return {name: item for name, item in inventory["files"].items()
                if not name.startswith("docs/") and name not in {
                    ".github/workflows/ci.yml",
                    "scripts/refresh_examples.py",
                    "tests/unit/test_onboarding_examples.py",
                    "examples/README.md",
                }}
    protected_before = retained_inputs(before)
    protected_prior = retained_inputs(prior)
    assert protected_before == protected_prior, "Tool-only exception cannot cover files beyond the exact four reviewed CI/tool/test/index paths and docs"
    protected_digest = digest(json.dumps(protected_before, sort_keys=True, separators=(",", ":")).encode())
    write("product-input-identity.json", {
        "schema_version": 1,
        "current_quality_commit": commit,
        "prior_product_candidate": prior["commit"],
        "prior_input_inventory_source_sha256": digest(prior_path.read_bytes()),
        "protected_input_count": len(protected_before),
        "protected_input_inventory_sha256": protected_digest,
        "all_protected_file_sets_and_bytes_identical": True,
        "scope": "Every tracked input except docs/ and the exact four authorized CI/tool/test/index paths matches the previous product candidate. Other src/tests/scripts/example inputs, schemas, dependencies, package metadata and release workflow remain identical.",
        "authorized_tool_only_paths": [
            ".github/workflows/ci.yml", "scripts/refresh_examples.py",
            "tests/unit/test_onboarding_examples.py", "examples/README.md",
        ],
        "previous_product_evidence": "Retained at b86c014; not relabeled as rebuilt or rerun for this CI-only commit.",
    })
    plan = command_plan(out, uv, commit)
    write(
        "plan.json",
        {
            "schema_version": 1,
            "candidate_commit": commit,
            "commands": [{"label": label, "argv": argv} for label, argv in plan],
            "command_count": len(plan),
            "status": "PREPARED_FOR_THIS_ATTEMPT",
        },
    )
    commands: list[dict[str, Any]] = []
    failed: str | None = None
    failure_kind: str | None = None

    def execute(label: str, argv: list[str]) -> int:
        origin_started = time.monotonic()
        origin = subprocess.run(
            [str(PYTHON), "-c", ORIGIN_CODE, str(ROOT)],
            cwd=ROOT,
            env=env,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
        )
        origin_elapsed = round(time.monotonic() - origin_started, 6)
        origin_name = label + "-origin.log"
        (out / "raw" / origin_name).write_bytes(origin.stdout)
        write(
            "active-command.json",
            {"schema_version": 1, "label": label, "argv": argv, "status": "STARTING"},
        )
        if origin.returncode:
            write(
                "origin-failure.json",
                {
                    "schema_version": 1,
                    "label": label,
                    "exit_code": origin.returncode,
                    "elapsed_seconds": origin_elapsed,
                    "raw_log_sha256": digest(origin.stdout),
                },
            )
            raise RuntimeError("candidate module/environment/version origin guard failed")
        assert json.loads(origin.stdout)["verified"] is True
        print("START " + label, flush=True)
        started = time.monotonic()
        result = subprocess.run(
            argv, cwd=ROOT, env=env, stdout=subprocess.PIPE, stderr=subprocess.STDOUT
        )
        elapsed = round(time.monotonic() - started, 6)
        name = label + ".log"
        (out / "raw" / name).write_bytes(result.stdout)
        shared = safe(result.stdout.decode("utf-8", errors="strict")).encode()
        (out / "logs" / name).write_bytes(shared)
        commands.append(
            {
                "label": label,
                "argv": argv,
                "cwd": "<CANDIDATE>",
                "exit_code": result.returncode,
                "elapsed_seconds": elapsed,
                "raw_log": "raw/" + name,
                "raw_log_sha256": digest(result.stdout),
                "shareable_log": "logs/" + name,
                "shareable_log_sha256": digest(shared),
                "origin_check": {
                    "exit_code": 0,
                    "elapsed_seconds": origin_elapsed,
                    "raw_log": "raw/" + origin_name,
                    "raw_log_sha256": digest(origin.stdout),
                },
                "candidate_input_inventory_sha256": before["inventory_sha256"],
            }
        )
        write(
            "commands.json",
            {
                "schema_version": 1,
                "task": "S4-A CI-only example-loop follow-up final quality",
                "candidate_commit": commit,
                "candidate_input_inventory_sha256": before["inventory_sha256"],
                "commands": commands,
            },
        )
        write(
            "active-command.json",
            {
                "schema_version": 1,
                "label": label,
                "status": "COMPLETE",
                "exit_code": result.returncode,
            },
        )
        print(f"DONE {label} exit={result.returncode} elapsed={elapsed}", flush=True)
        if result.returncode:
            print(safe(result.stdout.decode("utf-8", errors="strict"))[-16000:], flush=True)
        return result.returncode

    try:
        for label, argv in plan:
            if execute(label, argv):
                failed, failure_kind = label, "quality_command_nonzero"
                break
    except Exception as exc:
        failed, failure_kind = label, "orchestration_" + type(exc).__name__
    try:
        after = input_inventory(commit)
        write("inputs-after.json", after)
        unchanged = (
            after == before and git("status", "--porcelain=v1", "--untracked-files=all") == b""
        )
    except Exception as exc:
        unchanged = False
        write("inputs-after-failure.json", {"schema_version": 1, "error_type": type(exc).__name__})
    harness_unchanged = digest(Path(__file__).read_bytes()) == harness_sha
    completed = len(commands) == len(plan)
    passed = failed is None and completed and unchanged and harness_unchanged
    summary = {
        "schema_version": 1,
        "task": "S4-A CI-only example-loop follow-up final quality",
        "candidate_commit": commit,
        "all_passed": passed,
        "completed": completed,
        "failed_command": failed,
        "failure_kind": failure_kind,
        "command_count": len(commands),
        "planned_command_count": len(plan),
        "origin_check_count": len(commands) + int((out / "origin-failure.json").exists()),
        "candidate_input_inventory_sha256": before["inventory_sha256"],
        "inputs_byte_identical_before_after": unchanged,
        "orchestrator_sha256": harness_sha,
        "orchestrator_unchanged": harness_unchanged,
        "git_compatibility_shim_sha256": EXPECTED_SHIM,
        "summed_check_elapsed_seconds": round(
            sum(entry["elapsed_seconds"] for entry in commands), 6
        ),
        "summed_origin_guard_elapsed_seconds": round(
            sum(entry["origin_check"]["elapsed_seconds"] for entry in commands), 6
        ),
        "wall_seconds": round(time.monotonic() - started_all, 6),
        "environment_synced": False,
        "resource_commands_executed": 0,
        "standing_five_project_gate_a_commands_executed": 0,
        "product_gate_a_status": "NOT_APPLICABLE_CI_ONLY",
        "resource_gate_b_status": "NOT_APPLICABLE_CI_ONLY",
        "wheel_build_installation_status": "NOT_RERUN_PRODUCT_INPUTS_IDENTICAL",
        "prior_product_candidate": prior["commit"],
        "protected_input_inventory_sha256": protected_digest,
        "scope_guard": "All tracked inputs except docs and four authorized CI/tool/test/index paths byte-identical to prior product candidate.",
        "cross_platform_ci_status": "NOT_RUN_BY_THIS_LOCAL_RUNNER",
        "published_action_step": "CI_ONLY_NOT_RUN_LOCALLY",
        "validation_tools": (
            "Worktree locked environment; source origin and installed 0.7.0 "
            "metadata checked before each command."
        ),
        "cadence_basis": "docs/plan/validation-workflow-2026-10-13.md sections 3-5",
        "next_step": (
            "After PASS, use CI wiring positive/negative receipts and verify remote CI for the actual pushed HEAD. "
            "Product Gate A/B and candidate-wheel evidence stay bound to the unchanged b86c014 product candidate."
        ),
    }
    write("summary.json", summary)
    print(json.dumps(summary, sort_keys=True), flush=True)
    return 0 if passed else 1


if __name__ == "__main__":
    sys.exit(main())
